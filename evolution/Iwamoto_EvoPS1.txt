#create a hill climbing algorithm to find the maximum likelihood estimate (MLE)

function findML(Natural n, Probability, Real stepSize){
    like = calcLike(n, p, 1)
    while (calcLike(n, Probability+stepSize, 1) > like){
        p = p + stepSize
        like = calcLike(n, p, 1)
    }
    while (calcLike(n, p-stepSize, 1) > like){
        p = p - stepSize
        like = calcLike(n, p, 1)
    }
    if(stepSize > 0.0001){
        p = findML(n, p, stepSize/2.0)
    }
    return p
}
